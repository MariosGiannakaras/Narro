from pathlib import Path
import subprocess,time,datetime,json,shutil,sys,ctypes
r=Path(__file__).parent;name=sys.argv[1];d=r/name;d.mkdir(exist_ok=False)
ctypes.windll.user32.SetProcessDPIAware()
u=ctypes.windll.user32
box={k:u.GetSystemMetrics(i) for k,i in [('x',76),('y',77),('width',78),('height',79)]}
with (d/'stderr.txt').open('w') as err:
 p=subprocess.Popen([shutil.which('ffmpeg'),'-hide_banner','-loglevel','warning','-f','gdigrab','-framerate','30','-offset_x',str(box['x']),'-offset_y',str(box['y']),'-video_size',str(box['width'])+'x'+str(box['height']),'-draw_mouse','1','-i','desktop','-t','300','-c:v','libx264','-preset','ultrafast','-crf','18','-pix_fmt','yuv420p',str(d/'native-original.mkv')],stdin=subprocess.PIPE,stderr=err,stdout=subprocess.DEVNULL)
 (d/'start.json').write_text(json.dumps({'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'pid':p.pid,'physicalVirtualDesktop':box}))
 end=time.monotonic()+295
 while p.poll() is None and time.monotonic()<end and not (r/(name+'.stop')).exists():time.sleep(.25)
 if p.poll() is None:p.communicate(b'q\n',timeout=10)
 (d/'stop.json').write_text(json.dumps({'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'exit':p.returncode}))
