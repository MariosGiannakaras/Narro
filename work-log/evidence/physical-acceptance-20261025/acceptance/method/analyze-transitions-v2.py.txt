from pathlib import Path
import json,datetime,subprocess,math,re
from PIL import Image,ImageDraw
r=Path(__file__).parent;out=r/'analysis-v2';out.mkdir(exist_ok=True)
def read(p):
 b=p.read_bytes();return json.loads(b.decode('utf-16' if b[:2] in [b'\xff\xfe',b'\xfe\xff'] else 'utf-8-sig'))
def dt(x):return datetime.datetime.fromisoformat(x.replace('Z','+00:00'))
allcases=[]
for v in ['A','B']:
 events=[json.loads(x) for x in (r/('native-traces-'+v)/'focus-transition-trace.jsonl').read_text(encoding='utf-8-sig').splitlines()]
 begins=[x for x in events if x['stage']=='native.begin'];assert len(begins)==8
 for index,b in enumerate(begins):
  token=b['rasterToken'];e=next(x for x in events if x['rasterToken']==token and x['stage']=='native.committed');mode='normal' if index<4 else 'reduced';segment=v+'-'+mode;name=segment+'-'+str(index%4+1);start=read(r/segment/'start.json');frames=read(r/(segment+'-frames.json'));before=[f for f in frames if f['captureQpcAfter']<b['qpcTicks']];during=[f for f in frames if f['captureQpcBefore']<=e['qpcTicks'] and f['captureQpcAfter']>=b['qpcTicks']];after=[f for f in frames if f['captureQpcBefore']>e['qpcTicks']]
  begin=(dt(b['utc'])-dt(start['utc'])).total_seconds();end=(dt(e['utc'])-dt(start['utc'])).total_seconds();seek=max(0,begin-.6);count=math.ceil((end-begin+1.6)*60);folder=out/name;folder.mkdir(exist_ok=True)
  cmd=['ffmpeg','-hide_banner','-loglevel','info','-ss',str(seek),'-copyts','-i',str(r/segment/'native-original.mkv'),'-filter_complex','[0:v]split=2[a][b];[a]crop=340:700:0:0[l];[b]crop=340:700:2220:0[rr];[l][rr]hstack,showinfo','-frames:v',str(count),'-fps_mode','passthrough',str(folder/'frame-%03d.png')]
  result=subprocess.run(cmd,stdout=subprocess.DEVNULL,stderr=subprocess.PIPE);(folder/'extraction.txt').write_bytes(result.stderr);assert result.returncode==0
  pts=[float(x) for x in re.findall(rb'pts_time:([\d.]+)',result.stderr)];files=sorted(folder.glob('frame-*.png'));assert len(files)>0;pts=pts[:len(files)]
  import hashlib
  unique=[];previous=None
  for ix,p in enumerate(files):
   digest=hashlib.sha256(Image.open(p).tobytes()).hexdigest()
   if digest!=previous:unique.append((ix,p));previous=digest
  for part in range(math.ceil(len(unique)/24)):
   selected=unique[part*24:(part+1)*24];sheet=Image.new('RGB',(1360,6*372),(30,30,30));draw=ImageDraw.Draw(sheet)
   for i,(n,p) in enumerate(selected):
    im=Image.open(p).resize((340,350));x=(i%4)*340;y=(i//4)*372;sheet.paste(im,(x,y));draw.text((x+3,y+351),f'{name} #{n} PTS{pts[n]:.3f}',fill='white')
   sheet.save(out/(name+'-sheet-'+str(part+1)+'.jpg'),quality=94)
  row={'case':name,'from':b['details']['data']['from'],'to':b['details']['data']['to'],'begin':b,'commit':e,'videoBeginApprox':begin,'videoCommitApprox':end,'qpcBefore':len(before),'qpcDuring':len(during),'qpcAfter':len(after),'qpcFirstAfter':after[0] if after else None,'qpcWindowOrigins':sorted(set((f['native']['window']['x'],f['native']['window']['y']) for f in frames)),'extractedFrames':len(files),'firstPTS':pts[0],'lastPTS':pts[-1]};allcases.append(row);print(name,'QPC',len(before),len(during),len(after),'video',len(files),flush=True)
(out/'cases.json').write_text(json.dumps(allcases,indent=2),encoding='utf-8')
