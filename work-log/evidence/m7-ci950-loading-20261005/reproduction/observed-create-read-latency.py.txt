import argparse,ctypes,datetime,hashlib,json,os,pathlib,sqlite3,time
parser=argparse.ArgumentParser()
parser.add_argument('--window',type=int,required=True)
parser.add_argument('--sha256',required=True)
parser.add_argument('--output',type=pathlib.Path,required=True)
parser.add_argument('--finish',choices=['escape','ready','settle'],default='escape')
parser.add_argument('--hold-seconds',type=float,default=1.8)
a=parser.parse_args();a.output.parent.mkdir(parents=True,exist_ok=True)
assert 1.5<=a.hold_seconds<=5,'Bounded latency required'
user=ctypes.WinDLL('user32',use_last_error=True);kernel=ctypes.WinDLL('kernel32',use_last_error=True)
user.GetForegroundWindow.restype=ctypes.c_void_p
user.GetWindowThreadProcessId.argtypes=[ctypes.c_void_p,ctypes.POINTER(ctypes.c_ulong)]
kernel.OpenProcess.argtypes=[ctypes.c_ulong,ctypes.c_int,ctypes.c_ulong];kernel.OpenProcess.restype=ctypes.c_void_p
kernel.QueryFullProcessImageNameW.argtypes=[ctypes.c_void_p,ctypes.c_ulong,ctypes.c_wchar_p,ctypes.POINTER(ctypes.c_ulong)]
kernel.CloseHandle.argtypes=[ctypes.c_void_p]
assert user.GetForegroundWindow()==a.window,'Observed target is not foreground'
owner=ctypes.c_ulong();user.GetWindowThreadProcessId(a.window,ctypes.byref(owner))
handle=kernel.OpenProcess(0x1000,False,owner.value);assert handle
try:
    buffer=ctypes.create_unicode_buffer(32768);size=ctypes.c_ulong(len(buffer))
    assert kernel.QueryFullProcessImageNameW(handle,0,buffer,ctypes.byref(size))
    exe=pathlib.Path(buffer.value)
    assert exe.name=='narro-m7-validation.exe' and hashlib.file_digest(exe.open('rb'),'sha256').hexdigest()==a.sha256
finally:kernel.CloseHandle(handle)
db=pathlib.Path(os.environ['APPDATA'])/'com.mariosg.Narro/narro.db'
connection=sqlite3.connect(db,timeout=0.25,isolation_level=None)
assert connection.execute('PRAGMA journal_mode').fetchone()[0]=='delete','Delay method requires current rollback journal mode'

def log(action,**values):
    row=dict(utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),action=action,windowId=a.window,processId=owner.value,exeSha256=a.sha256,**values)
    with a.output.open('a',encoding='utf-8') as f:f.write(json.dumps(row,ensure_ascii=False)+'\n')
    print(json.dumps(row,ensure_ascii=False),flush=True)
def chord(scans,label):
    assert user.GetForegroundWindow()==a.window,'Foreground changed during observed input'
    try:
        for scan in scans:user.keybd_event(0,scan,8,0);time.sleep(0.02)
        time.sleep(0.04)
    finally:
        for scan in reversed(scans):user.keybd_event(0,scan,10,0);time.sleep(0.02)
    log('actual-scancode-input',chord=label,scancodes=scans)

connection.execute('BEGIN EXCLUSIVE')
log('read-latency-lock-acquired',method='BEGIN EXCLUSIVE; no SQL data statements; rollback/connection close release',checkpoint=checkpoint)
try:
    chord([0x1d,0x38,0x14],'Ctrl+Alt+T')

        chord([0x0f],'Tab');time.sleep(0.2)

    remaining=a.hold_seconds-(time.monotonic()-started)

    connection.rollback();connection.close()
    log('read-latency-lock-released',heldSeconds=round(time.monotonic()-started,4),finish=a.finish)
